function recalcRow(month) {
  const row = document.querySelector(`tr[data-month="${month}"]`);
  const basic     = parseFloat(row.querySelector('.p9a-basic').value) || 0;
  const benefits  = parseFloat(row.querySelector('.p9a-benefits').value) || 0;
  const ownerInt  = parseFloat(row.querySelector('.p9a-owner-int').value) || 0;
  const pension   = parseFloat(row.querySelector('.p9a-pension').value) || 0;

  const gross = basic + benefits;
  const nssf  = Math.min(gross * 0.06, 2160); // simplified NSSF tier
  const chargeable = Math.max(0, gross - nssf - ownerInt - pension);

  // PAYE calc (same bands as payslip prompt)
  const paye = calcPAYE(chargeable); // reuse from shared calc utility

  const shif = gross * 0.0275;
  const hl   = gross * 0.015;
  const net  = gross - paye - nssf - shif - hl;

  // Fill calculated cells
  row.querySelector('.p9a-gross').textContent          = fmt(gross);
  row.querySelector('.p9a-nssf').textContent           = fmt(nssf);
  row.querySelector('.p9a-chargeable').textContent     = fmt(chargeable);
  row.querySelector('.p9a-tax-chargeable').textContent = fmt(paye + 2400); // gross tax before relief
  row.querySelector('.p9a-personal-relief').textContent= fmt(2400);
  row.querySelector('.p9a-paye').textContent           = fmt(paye);
  row.querySelector('.p9a-shif').textContent           = fmt(shif);
  row.querySelector('.p9a-hl').textContent             = fmt(hl);
  row.querySelector('.p9a-net').textContent            = fmt(net);

  updateP9ATotals();
}

## STEP 2 — Auto-Populate from Payroll History

When an employee + tax year is selected, auto-fill the monthly gross from `payroll_history`:

```js
async function autoPopulateFromHistory(employeeId, taxYear) {
  try {
    const { data, error } = await supabaseClient
      .from('payroll_history')
      .select('pay_period, gross_pay, paye, nssf, shif, housing_levy, net_pay')
      .eq('employee_id', employeeId)
      .like('pay_period', `${taxYear}-%`) // e.g. '2025-%'
      .order('pay_period', { ascending: true });

    if (error) throw error;
    if (!data || data.length === 0) {
      showNotice('No payroll history found for this employee/year. Enter manually.');
      return;
    }

    // Map each record to its month row
    data.forEach(record => {
      const month = parseInt(record.pay_period.split('-')[1]); // extract month number
      const row = document.querySelector(`tr[data-month="${month}"]`);
      if (row) {
        row.querySelector('.p9a-basic').value = record.gross_pay || 0;
        recalcRow(month);
      }
    });

    showSuccess('✅ P9A auto-populated from payroll history.');
  } catch (err) {
    console.error('Auto-populate failed:', err);
    showError('Could not load payroll history. Please enter values manually.');
  }
}
```

## STEP 4 — PDF Export (P9A layout)

The P9A PDF must look like the official KRA form. Use html2pdf.js (already added in Prompt 02):

```js
function generateP9APDF() {
  const employeeName = document.getElementById('employee-select')?.selectedOptions[0]?.text
    || userProfile?.full_name || 'Employee';
  const taxYear = document.getElementById('tax-year')?.value || new Date().getFullYear() - 1;

  // Build a clean printable version of the P9A (hide input controls, show only data)
  const printTarget = document.getElementById('p9a-print-view');
  // ... ensure p9a-print-view mirrors the table with static text cells

  const opt = {
    margin:      [8, 5, 8, 5],
    filename:    `P9A-${employeeName.replace(/\s+/g,'-')}-${taxYear}.pdf`,
    image:       { type: 'jpeg', quality: 0.98 },
    html2canvas: { scale: 2, useCORS: true },
    jsPDF:       { unit: 'mm', format: 'a4', orientation: 'landscape' } // P9A is landscape
  };
  html2pdf().set(opt).from(printTarget).save();
}
```

> **Important:** P9A table is wide — use `orientation: 'landscape'` for A4 PDF.
> Add `@media print { .p9a-controls { display:none } }` so print view is clean.

## STEP 5 — Fix Supabase .catch errors (same as Prompt 01)

Apply the `await`/`try-catch` pattern to ALL Supabase calls in this file:
- Save Draft
- Load Past Records
- Auto-populate query
- Any `.insert`, `.update`, `.upsert`, `.select` calls

```js
// ❌ WRONG
supabaseClient.from('p9a_records').insert(record).catch(console.error);

// ✅ CORRECT
async function saveP9ADraft(record) {
  try {
    const { error } = await supabaseClient.from('p9a_records').insert(record);
    if (error) throw error;
    showSuccess('Draft saved.');
  } catch (err) {
    console.error('Save failed:', err);
    showError('Failed to save draft.');
  }
}
```
